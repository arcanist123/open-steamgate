// OSG's pinned extraction adapter. Policy lives in osg/gogen/cdscheck.
package @package@

type OSGExpression struct { Kind, Text string; Children []OSGExpression }
type OSGFile struct { Filename, Raw string }
type OSGObject struct { Found bool; Files []OSGFile; Tree *OSGExpression; SQLView string; Components, Keys []string }
func osgExpression(n @ExpressionNode@) OSGExpression {
 out:=OSGExpression{Kind:classOf(n.@get@()).Name.String(), Text:n.@concatTokens@().String()}
 children:=n.@getChildren@()
 for i:=int32(0);i<libraryLength(children);i++ {
  if child,ok:=libraryUnwrap(libraryAt(children,i)).(@ExpressionNode@);ok { out.Children=append(out.Children,osgExpression(child)) }
 }
 return out
}
// OSGInspect copies original parser facts under the library lock, without
// exposing generated types. Empty component lists match optional getComponents.
func (c *Checker) OSGInspect(typ,name string) OSGObject {
 libraryMu.Lock(); defer libraryMu.Unlock(); c.require()
 c.registry.@parse@()
 obj:=c.registry.@getObject@(present(str(typ)),present(str(name)))
 if obj==nil { return OSGObject{} }
 out:=OSGObject{Found:true}
 files:=obj.@getFiles@()
 for i:=int32(0);i<libraryLength(files);i++ { f:=libraryUnwrap(libraryAt(files,i)).(@File@); out.Files=append(out.Files,OSGFile{Filename:f.@getFilename@().String(),Raw:f.@getRaw@().String()}) }
 var parsed @AbstractType@
 switch o:=obj.(type) {
 case @DataDefinition@:
  if tree:=o.@getTree@();tree!=nil { n:=osgExpression(tree);out.Tree=&n }
  if sql:=o.@getSQLViewName@();sql.Has { out.SQLView=sql.Value.String() }
 case @Table@:
  parsed=o.@parseType@(c.registry,optional[jsString]{},optional[jsString]{})
  for _,k:=range o.@listKeys@(c.registry).Items { out.Keys=append(out.Keys,k.String()) }
 case @View@:
  parsed=o.@parseType@(c.registry,optional[jsString]{},optional[jsString]{})
  for _,k:=range o.@listKeys@().Items { out.Keys=append(out.Keys,k.String()) }
 }
 if structure,ok:=parsed.(@StructureType@);ok { for _,v:=range structure.@getComponents@().Items { component:=libraryUnwrap(v).(@Component@);out.Components=append(out.Components,component.@name@.String()) } }
 return out
}
